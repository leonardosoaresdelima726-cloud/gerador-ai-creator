function avancar(passo){

  document.querySelectorAll('.step-panel').forEach(p=>{p.classList.remove('active');p.classList.add('hidden');});

  document.getElementById(`step${passo}`).classList.remove('hidden');

  setTimeout(()=>document.getElementById(`step${passo}`).classList.add('active'),10);

  atualizarBarra(passo);

}

function voltar(passo){

  document.querySelectorAll('.step-panel').forEach(p=>{p.classList.remove('active');p.classList.add('hidden');});

  document.getElementById(`step${passo}`).classList.remove('hidden');

  setTimeout(()=>document.getElementById(`step${passo}`).classList.add('active'),10);

  atualizarBarra(passo);

}

function atualizarBarra(passo){

  const porcentagens={1:25,2:50,3:75,4:100};

  document.getElementById('progressBar').style.width=porcentagens[passo]+'%';

  document.getElementById('progressPercent').textContent=porcentagens[passo]+'%';

  const textos={1:'Passo 1 de 4',2:'Passo 2 de 4',3:'Passo 3 de 4',4:'Concluído!'};

  document.getElementById('progressText').textContent=textos[passo];

}
